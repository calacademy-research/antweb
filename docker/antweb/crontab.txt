#27 3 * * * root sh /antweb/deploy/bin/reboot.sh # Restart at 3:27 am every day.
# The jobs run as root. Their logs are in web/log, which tomcat owns, so they are
# written through tee running as tomcat: root never opens a file there (tomcat
# could have replaced it with a symlink to, say, a database dump).
0 21 * * * root /antweb/deploy/bin/admin.sh | setpriv --reuid=tomcat --regid=antweb --init-groups tee /data/antweb/web/log/adminTask.log > /dev/null
0 23 1 * * root /antweb/deploy/bin/db-monthly-dump.sh
50 3 * * 4 root /antweb/deploy/bin/taxonSetBackup.sh | setpriv --reuid=tomcat --regid=antweb --init-groups tee -a /data/antweb/web/log/taxonSetBackup.log > /dev/null
# Log rotation (docker/antweb/logrotate.conf), as tomcat; its own report goes to log/logrotate.log.
40 3 * * * root setpriv --reuid=tomcat --regid=antweb --init-groups /usr/sbin/logrotate --state /usr/local/antweb/log/.logrotate.state --log /usr/local/antweb/log/logrotate.log /etc/antweb-logrotate.conf
